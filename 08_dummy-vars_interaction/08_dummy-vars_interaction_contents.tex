\section{Qualitative regressors}

    \begin{itemize}
        \item So far every regressor has been quantitative: speed, temperature, delivery volume.
        \item Many important predictors are qualitative (categorical): sex, region, treatment group, machine type.

\section{Indicator variables}


\begin{frame}{Indicator (dummy) variable}
    \begin{itemize}
        \item An indicator variable encodes membership in a category with the values $0$ and $1$.
        \item For tool type:
    \end{itemize}
    \[
        x_2 =
        \begin{cases}
            0 & \text{if the observation is from tool type A},\\
            1 & \text{if the observation is from tool type B}.
        \end{cases}
    \]
    \begin{itemize}
        \item Type A is coded $0$: it is the \textbf{reference} (baseline) category.
    \end{itemize}
\end{frame}

\begin{frame}{Model with one quantitative regressor and one indicator}
    \begin{itemize}
        \item Regress $y$ on speed $x_1$ and the indicator $x_2$:
    \end{itemize}
    \[
        y_i = \beta_0 + \beta_1 x_{i1} + \beta_2 x_{i2} + \varepsilon_i, \qquad \varepsilon_i \sim N(0, \sigma^2).
    \]
    \begin{itemize}
        \item This is an ordinary multiple linear regression; $\bX$ has columns $\mathbf{1}$, $\bx_1$, $\bx_2$, and OLS applies unchanged.
        \item Only the interpretation of the coefficient on $x_2$ is new.
    \end{itemize}
\end{frame}


\begin{frame}{Two regression lines from one model}
    \begin{itemize}
        \item Setting $x_2 = 0$ and $x_2 = 1$ gives the mean response for each tool type:
    \end{itemize}
    \begin{align*}
        \text{Type A}\colon \quad \E[y \mid x_1, x_2 = 0] &= \beta_0 + \beta_1 x_1, \\
        \text{Type B}\colon \quad \E[y \mid x_1, x_2 = 1] &= (\beta_0 + \beta_2) + \beta_1 x_1.
    \end{align*}
    \begin{itemize}
        \item The lines have the same slope $\beta_1$ and intercepts that differ by $\beta_2$: the lines are \textbf{parallel}.
        \item $\beta_2$ is the difference in mean tool life between type B and type A at any fixed speed.
    \end{itemize}
\end{frame}

\section{Inference on the indicator coefficient}


\begin{frame}{Full and reduced models}
    \begin{itemize}
        \item A hypothesis $H_0$ about the coefficients, e.g.\ that some of them are zero, defines a simpler model.
        \item \textbf{Full model (FM):} all terms; $p$ parameters.
        \item \textbf{Reduced model (RM):} the FM with $H_0$ imposed; $p - q$ parameters, where $q$ is the number of restrictions in $H_0$.
        \item The RM is \emph{nested} in the FM: every RM is a special case of the FM.
    \end{itemize}
    \vspace{0.5em}
    Example: $H_0\colon \beta_2 = 0$ ($q = 1$).
    \begin{align*}
        \FM\colon\quad y &= \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \varepsilon, \\
        \RM\colon\quad y &= \beta_0 + \beta_1 x_1 + \varepsilon .
    \end{align*}
\end{frame}


\begin{frame}{Extra sum of squares}
    \begin{itemize}
        \item Least squares searches over all coefficient values allowed by the model; the FM allows more, so
    \end{itemize}
    \[
        \SSRes(\RM) \ge \SSRes(\FM).
    \]
    \begin{itemize}
        \item The difference is the \textbf{extra sum of squares}: the increase in the residual sum of squares caused by imposing $H_0$,
    \end{itemize}
    \[
        \SSRes(\RM) - \SSRes(\FM) \ge 0 .
    \]
    \begin{itemize}
        \item Small increase: the dropped terms add little, and $H_0$ is plausible. Large increase: the dropped terms matter.
        \item \textbf{Question:} how large is ``large''?
        % Answer: compare it to the noise level, i.e.\ to an estimate of sigma^2 from the full model, MSRes(FM) = SSRes(FM)/(n-p); and divide by q to get the increase per restriction.
    \end{itemize}
\end{frame}


\begin{frame}{$F$-test for nested models}
    \begin{itemize}
        \item Under $H_0$, with $n$ observations, $p$ parameters in the FM and $q$ restrictions:
    \end{itemize}
    \[
        F_0 = \frac{\left[\SSRes(\RM) - \SSRes(\FM)\right] / q}{\SSRes(\FM) / (n - p)} \sim F_{q,\, n-p}.
    \]
    \begin{itemize}
        \item Numerator: increase in $\SSRes$ per restriction. Denominator: $\MSRes$ of the FM, an estimate of $\sigma^2$.
        \item We use this distributional result here; Lecture 9 derives it.
        \item Reject $H_0$ at level $\alpha$ if $F_0 > F_{\alpha,\, q,\, n-p}$.
        \item For $q = 1$ the test is equivalent to the two-sided $t$-test of the coefficient, with $F_0 = t_0^2$.
    \end{itemize}
\end{frame}


\begin{frame}{Inference on the indicator coefficient}
    \begin{itemize}
        \item $H_0\colon \beta_2 = 0$ (the two tool types have the same mean life at every speed) is tested with the usual $t$-test; here $t_0 = 11.04$ on $17$ df.
        \item Equivalently, use the $F$-test with FM: $y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \varepsilon$ and RM: $y = \beta_0 + \beta_1 x_1 + \varepsilon$ (speed only); here $q = 1$, $n - p = 17$:
    \end{itemize}
    \begin{align*}
        F_0 &= \frac{\left[\SSRes(\RM) - \SSRes(\FM)\right] / 1}{\SSRes(\FM) / 17}
        = \frac{(1282.08 - 157.05)/1}{157.05/17} = 121.8, \\
        F_0 &= t_0^2 .
    \end{align*}
    \begin{itemize}
        \item Reject $H_0$: tool type is strongly associated with tool life after adjusting for speed.
    \end{itemize}
\end{frame}


\begin{frame}{Special case: indicator only = two-sample $t$-test}
    \begin{itemize}
        \item Drop speed and regress $y$ on the indicator alone: $y = \beta_0 + \beta_2 x_2 + \varepsilon$.
        \item The group means are $\E[y \mid x_2 = 0] = \beta_0$ and $\E[y \mid x_2 = 1] = \beta_0 + \beta_2$, and OLS gives the sample means:
    \end{itemize}
    \[
        \hat{\beta}_0 = \bar{y}_A = 17.11, \qquad \hat{\beta}_2 = \bar{y}_B - \bar{y}_A = 31.93 - 17.11 = 14.82 .
    \]
    \begin{itemize}
        \item $t_0 = \hat{\beta}_2 / \se(\hat{\beta}_2) = 14.82 / 2.30 = 6.44$ on $n - 2 = 18$ df: exactly the pooled-variance two-sample $t$-test of $\mu_A = \mu_B$.
        \item With $m$ levels and $m - 1$ indicators, the overall $F$-test of the regression is the one-way ANOVA $F$-test.
        \item Adding speed raises $t_0$ for tool type from $6.44$ to $11.04$: speed explains part of the noise, so $\hat{\sigma}$ falls from $5.15$ to $3.04$.
    \end{itemize}
\end{frame}


\section{Factors with more than two levels}


\begin{frame}{Categorical regressor with $m$ levels}
    \begin{itemize}
        \item A categorical regressor is called a \textbf{factor}; its categories are its \textbf{levels} (tool type is a factor with levels A and B).
        \item We write $m$ for the number of levels ($k$ remains the number of regressors, $p = k + 1$).
        \item A factor with $m$ levels is represented by $m-1$ indicator variables.
        \item The level without an indicator is the reference level.
        \item Suppose there were three tool types A, B, C (hypothetical extension), with A as reference:
    \end{itemize}
    \[
        x_2 =
        \begin{cases}
            1 & \text{if type B},\\
            0 & \text{otherwise},
        \end{cases}
        \qquad
        x_3 =
        \begin{cases}
            1 & \text{if type C},\\
            0 & \text{otherwise}.
        \end{cases}
    \]
\end{frame}


\begin{frame}{Coding and interpretation for $m = 3$}
    \[
        y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \beta_3 x_3 + \varepsilon
    \]
    \centering
    \small
    \begin{tabular}{lrrl}
        \toprule
        Type & $x_2$ & $x_3$ & $\E[y \mid x_1, x_2, x_3]$ \\
        \midrule
        A & 0 & 0 & $\beta_0 + \beta_1 x_1$ \\
        B & 1 & 0 & $(\beta_0 + \beta_2) + \beta_1 x_1$ \\
        C & 0 & 1 & $(\beta_0 + \beta_3) + \beta_1 x_1$ \\
        \bottomrule
    \end{tabular}

    \vspace{0.8em}
    \raggedright
    $\beta_2$ and $\beta_3$ are the differences of B and C from the reference A, at fixed $x_1$. The difference between B and C is $\beta_2 - \beta_3$.
\end{frame}


\begin{frame}{The dummy variable trap}
    \begin{itemize}
        \item Suppose we include all $m = 3$ indicators $\mathbf{d}_A, \mathbf{d}_B, \mathbf{d}_C$ together with an intercept, where $\mathbf{d}_B = \bx_2$, $\mathbf{d}_C = \bx_3$ and $\mathbf{d}_A$ is the indicator of type A.
        \item Every observation belongs to exactly one type, so
    \end{itemize}
    \begin{align*}
        \mathbf{1} &= \mathbf{d}_A + \mathbf{d}_B + \mathbf{d}_C, \\
        \rank(\bX) &= 4 < 5 \text{ columns}.
    \end{align*}
    \begin{itemize}
        \item $\bX$ lacks full column rank, so $\bX' \bX$ is singular (Lecture 4): the normal equations have no unique solution and $\bbeta$ is not \textbf{identifiable}, i.e.\ different coefficient vectors give the same $\E[\by] = \bX\bbeta$.
        \item Remedy: drop one indicator (use $m-1$), or drop the intercept (then each coefficient is a level-specific intercept).
    \end{itemize}
\end{frame}


\begin{frame}{Question: counting indicators}
    \begin{itemize}
        \item A factor has $4$ levels and the model includes an intercept. How many indicators are needed? How many if the intercept is removed, and how does the meaning of the coefficients change?
        % Answer: 3 indicators with the intercept. Without the intercept, 4 indicators; each coefficient is then the mean response for that level (at x = 0 for any quantitative regressors).
    \end{itemize}
\end{frame}


\begin{frame}{Choice of reference level}
    \begin{itemize}
        \item Changing the reference level changes the coefficients, not the model.
        \item With reference C, use the indicators $d_A$, $d_B$: $\ \E[y] = \beta_0' + \beta_1' x_1 + \beta_A' d_A + \beta_B' d_B$.
        \item Both codings describe the same three lines, so the mean of each type must agree:
    \end{itemize}
    \centering
    \small
    \begin{tabular}{llll}
        \toprule
        Type & Reference A & Reference C & Solve \\
        \midrule
        C & $(\beta_0 + \beta_3) + \beta_1 x_1$ & $\beta_0' + \beta_1' x_1$ & $\beta_0' = \beta_0 + \beta_3$, $\ \beta_1' = \beta_1$ \\
        A & $\beta_0 + \beta_1 x_1$ & $(\beta_0' + \beta_A') + \beta_1' x_1$ & $\beta_A' = \beta_0 - \beta_0' = -\beta_3$ \\
        B & $(\beta_0 + \beta_2) + \beta_1 x_1$ & $(\beta_0' + \beta_B') + \beta_1' x_1$ & $\beta_B' = \beta_2 - \beta_3$ \\
        \bottomrule
    \end{tabular}

    \vspace{0.8em}
    \raggedright
    \begin{itemize}
        \item The slope $\beta_1$ does not change; only the differences are measured from C instead of A.
    \end{itemize}
\end{frame}


\begin{frame}{Reference level: what changes and what does not}
    \begin{itemize}
        \item Coefficients of the indicators and the intercept change with the reference level; their $t$-tests compare each level with a different baseline.
        \item Fitted values, residuals, $\SSRes$, $R^2$ and the overall $F$-test of the factor are identical for every choice of reference level.
        \item Choose the reference level that makes the comparisons of interest direct (e.g.\ a control group).
    \end{itemize}
\end{frame}


\begin{frame}{Testing the factor as a whole}
    \begin{itemize}
        \item Individual $t$-tests compare one level with the reference only.
        \item To test whether the factor matters at all, $H_0\colon \beta_2 = \beta_3 = 0$, use the partial $F$-test:
    \end{itemize}
    \[
        F_0 = \frac{\left[\SSRes(\RM) - \SSRes(\FM)\right]/(m-1)}{\SSRes(\FM)/(n - p)} \sim F_{m-1,\, n-p} \quad \text{under } H_0,
    \]
    \begin{itemize}
        \item where the reduced model omits all $m-1$ indicators and $p$ is the number of parameters of the full model.
    \end{itemize}
\end{frame}


\begin{frame}{Indicator variables versus allocated codes}
    \begin{itemize}
        \item Tempting alternative: one regressor with codes A $= 1$, B $= 2$, C $= 3$ (allocated codes).
        \item This forces $\E[y]$ to change by the same amount $\beta$ from A to B and from B to C, as if the levels were equally spaced on a numeric scale.
        \item For \textbf{nominal} categories (no natural order, e.g.\ tool type or region) that ordering and spacing is arbitrary; recoding to A $= 1$, C $= 2$, B $= 3$ would change the fit.
        \item Indicator variables impose no such structure: each level gets its own mean shift.
        \item Allocated codes are defensible only when the levels are genuinely ordered and roughly equally spaced.
    \end{itemize}
\end{frame}


\section{Interaction terms}


\begin{frame}{Do both tool types respond to speed in the same way?}
    \begin{itemize}
        \item The parallel-lines model assumes that increasing speed shortens life by the same amount for tool types A and B.
        \item \textbf{Question:} how can we let the slope on $x_1$ differ between the two types?
        % Answer: add the product x1*x2 (interaction term); its coefficient is the difference in slopes.
    \end{itemize}
\end{frame}


\begin{frame}{Indicator $\times$ quantitative interaction}
    \begin{itemize}
        \item Add the product of the indicator and the quantitative regressor:
    \end{itemize}
    \[
        y_i = \beta_0 + \beta_1 x_{i1} + \beta_2 x_{i2} + \beta_{12}\, x_{i1} x_{i2} + \varepsilon_i .
    \]
    \begin{itemize}
        \item The subscript of $\beta_{12}$ (read ``beta one-two'', not ``beta twelve'') names the regressors whose product the term contains: $x_1$ and $x_2$. In general, the coefficient of $x_j x_l$ is written $\beta_{jl}$.
        \item The effect of $x_1$ now depends on the category, and the effect of the category depends on $x_1$.
    \end{itemize}
\end{frame}


\begin{frame}{Group-specific lines}
    \begin{align*}
        \text{Type A}\colon \quad \E[y \mid x_1, x_2 = 0] &= \beta_0 + \beta_1 x_1, \\
        \text{Type B}\colon \quad \E[y \mid x_1, x_2 = 1] &= (\beta_0 + \beta_2) + (\beta_1 + \beta_{12})\, x_1 .
    \end{align*}
    \begin{itemize}
        \item $\beta_2$: difference in intercepts; $\beta_{12}$: difference in slopes.
        \item The difference between the types at speed $x_1$ is $\beta_2 + \beta_{12} x_1$; it changes linearly with $x_1$.
    \end{itemize}
\end{frame}


\begin{frame}{Geometry: non-parallel lines}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth]
                    \draw[->] (0,0) -- (5,0) node[below] {\small $x_1$};
                    \draw[->] (0,0) -- (0,4.4) node[above] {\small $y$};
                    \draw[very thick, blue!70!black] (0,1.0) -- (4.5,2.0) node[right] {\scriptsize Type A};
                    \draw[very thick, orange!85!black] (0,2.0) -- (4.5,4.0) node[right] {\scriptsize Type B};
                    \node[anchor=east] at (0,1.0) {\small $\beta_0$};
                    \node[anchor=east] at (0,2.0) {\small $\beta_0 + \beta_2$};
                    \node[blue!70!black, anchor=north west] at (2.6,1.45) {\scriptsize slope $\beta_1$};
                    \node[orange!85!black, anchor=south east] at (3.7,3.1) {\scriptsize slope $\beta_1 + \beta_{12}$};
                \end{tikzpicture}
            \end{center}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Two free intercepts and two free slopes.
                \item The difference B $-$ A grows with $x_1$ when $\beta_{12} > 0$.
                \item If $\beta_{12} \ne 0$, the lines cross at $x_1 = -\beta_2 / \beta_{12}$ (if this lies in the range of the data, the ranking of the types reverses).
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

    \[

\begin{frame}{Testing the interaction}
    \begin{itemize}
        \item $H_0\colon \beta_{12} = 0$ (parallel lines): is the interaction term needed beyond $x_1$ and $x_2$? Compare nested models, with $q = 1$ and $n - p = 16$:
    \end{itemize}
    \begin{align*}
        \FM\colon\quad y &= \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \beta_{12} x_1 x_2 + \varepsilon, \\
        \RM\colon\quad y &= \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \varepsilon .
    \end{align*}
\end{frame}


\begin{frame}{Testing the interaction: result}
    \begin{align*}
        F_0 &= \frac{\left[\SSRes(\RM) - \SSRes(\FM)\right] / 1}{\SSRes(\FM) / 16}
        = \frac{(157.05 - 140.98)/1}{140.98/16} = 1.82, \\
        t_0 &= -1.35 \quad (t_0^2 = F_0), \qquad p\text{-value} = 0.196 .
    \end{align*}
    \begin{itemize}
        \item No evidence that the slopes differ: the parallel-lines model is adequate for these data.
        \item The estimated slopes differ by $0.0119$ per rpm, but with $n = 20$ the estimate is too imprecise to distinguish it from zero.
    \end{itemize}
\end{frame}


\begin{frame}{Testing sequence}
    \begin{itemize}
        \item Test the interaction first: the meaning of $\beta_2$ depends on whether $x_1 x_2$ is in the model, so the indicator is tested in the model that survives.
    \end{itemize}
    \vspace{0.3em}
    \centering
    \small
    \begin{tabular}{clllrl}
        \toprule
        Step & $H_0$ & Question & FM vs.\ RM & $F_0$ & $p$ \\
        \midrule
        1 & $\beta_{12} = 0$ & Parallel lines? & interaction vs.\ parallel & $1.82$ & $0.196$ \\
        2 & $\beta_2 = 0$ & Do they coincide? & parallel vs.\ speed only & $121.8$ & $< 0.001$ \\
        \bottomrule
    \end{tabular}

    \vspace{0.8em}
    \raggedright
    \begin{itemize}
        \item Step 1 does not reject, so $x_1 x_2$ is dropped and step 2 is carried out in the parallel-lines model.
        \item Step 2 rejects. Conclusion for tool life: keep $x_1$ and $x_2$, drop $x_1 x_2$.
    \end{itemize}
\end{frame}


\begin{frame}{One-step alternative: are the two lines identical?}
    \begin{itemize}
        \item Test both differences at once, $H_0\colon \beta_2 = \beta_{12} = 0$ ($q = 2$): FM is the interaction model, RM is the speed-only regression:
    \end{itemize}
    \[
        F_0 = \frac{(1282.08 - 140.98)/2}{140.98/16} = 64.8 \quad (p < 0.001).
    \]
    \begin{itemize}
        \item Reject: one common line is not adequate; tool type matters.
        \item Rejection does not say whether the intercepts, the slopes or both differ; the two-step sequence answers that.
    \end{itemize}
\end{frame}


\begin{frame}{Question: why not fit two separate regressions?}
    \begin{itemize}
        \item The interaction model gives one line for each tool type. Why not simply fit two regressions, one for each type?
        % Answer: the point estimates are identical (the interaction model is saturated in the two groups). Differences: (1) the single model pools the error variance across the groups (sigma-hat^2 = 140.98/16 on 16 df instead of two estimates on 8 df each); (2) differences in intercepts and slopes can be tested directly; (3) the parallel-lines and equal-lines models are available as special cases.
    \end{itemize}
\end{frame}


\begin{frame}{Indicator variables versus separate regressions}
    \begin{itemize}
        \item Separate fits (type A, type B) reproduce the interaction model coefficients:
    \end{itemize}
    \centering
    \small
    \begin{tabular}{lll}
        \toprule
        Fit & Intercept & Slope \\
        \midrule
        Type A alone & $32.775 = \hat{\beta}_0$ & $-0.0210 = \hat{\beta}_1$ \\
        Type B alone & $56.745 = \hat{\beta}_0 + \hat{\beta}_2$ & $-0.0329 = \hat{\beta}_1 + \hat{\beta}_{12}$ \\
        \bottomrule
    \end{tabular}

    \vspace{0.8em}
    \raggedright
    \begin{itemize}
        \item Only the single model allows direct tests of $\beta_2$, $\beta_{12}$, and of the special cases (parallel, equal lines).
    \end{itemize}
\end{frame}


\begin{frame}{Common versus separate error variance}
    \begin{itemize}
        \item The interaction model assumes a common $\sigma^2$ and estimates it from all $n$ observations; separate fits estimate one $\sigma^2$ per group.
    \end{itemize}
    \centering
    \small
    \begin{tabular}{lrrr}
        \toprule
        Fit & $\SSRes$ & df & $\hat{\sigma}^2 = \MSRes$ \\
        \midrule
        Type A alone & $47.57$ & $8$ & $5.95$ \\
        Type B alone & $93.40$ & $8$ & $11.68$ \\
        Interaction model (pooled) & $140.98$ & $16$ & $8.81$ \\
        \bottomrule
    \end{tabular}

    \vspace{0.8em}
    \raggedright
    \begin{itemize}
        \item Pooled $\SSRes$ = sum of the separate ones; its $16$ df (instead of $8$) give shorter intervals and more powerful tests, if the groups share one $\sigma^2$.
        \item The group estimates differ by a factor of about $2$; with $8$ df each this is within sampling variation (variance-ratio $F$-test, $p \approx 0.36$). Check with residual plots by group (Lecture 6).
    \end{itemize}
\end{frame}


\begin{frame}{Interaction between two quantitative regressors}
    \begin{itemize}
        \item Interaction is not limited to indicators:
    \end{itemize}
    \[
        y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \beta_{12} x_1 x_2 + \varepsilon .
    \]
    \begin{itemize}
        \item The effect on $\E[y]$ of a unit change in $x_1$ depends on the level of $x_2$:
    \end{itemize}
    \[
        \frac{\partial\, \E[y]}{\partial x_1} = \beta_1 + \beta_{12}\, x_2 ,
        \qquad
        \frac{\partial\, \E[y]}{\partial x_2} = \beta_2 + \beta_{12}\, x_1 .
    \]
    \begin{itemize}
        \item The mean response surface is no longer a plane; slices at fixed $x_2$ are lines with different slopes (this is the indicator case with $x_2 \in \{0, 1\}$).
    \end{itemize}
\end{frame}

\begin{frame}{Interpreting coefficients when an interaction is present}
    \begin{itemize}
        \item The coefficients of the single regressors ($\beta_1$, $\beta_2$) are called \textbf{main effects}, as opposed to the interaction $\beta_{12}$. With $x_1 x_2$ in the model, $\beta_1$ is the effect of $x_1$ only at $x_2 = 0$; it is not an overall effect of $x_1$ unless $x_2 = 0$ is meaningful.
        \item Centering the quantitative regressors at their means makes $\beta_1$ and $\beta_2$ interpretable at average values. It also reduces \textbf{collinearity}, i.e.\ strong correlation between $x_1 x_2$ and its components, which inflates standard errors (Lecture 4: nearly dependent columns make $\bX'\bX$ nearly singular; treated in detail in a later lecture).
    \end{itemize}
\end{frame}


\begin{frame}{Practical guidance on interactions}
    \begin{itemize}
        \item \textbf{Hierarchy principle:} whenever $x_1 x_2$ is in the model, keep the lower-order terms $x_1$ and $x_2$, even if their $t$-tests are not significant.
        \item Do not interpret interaction coefficients in isolation; plot the fitted lines at chosen values of the other regressor.
        \item Each additional interaction costs a degree of freedom; add them for substantive reasons, not by default.
    \end{itemize}
\end{frame}


\begin{frame}{Interaction with an $m$-level factor}
    \begin{itemize}
        \item For a three-level factor (indicators $x_2$, $x_3$) allow a separate slope for each level:
    \end{itemize}
    \[
        y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \beta_3 x_3 + \beta_{12} x_1 x_2 + \beta_{13} x_1 x_3 + \varepsilon .
    \]
    \begin{itemize}
        \item Slopes in $x_1$: $\beta_1$ (reference A), $\beta_1 + \beta_{12}$ (B), $\beta_1 + \beta_{13}$ (C).
        \item Test of equal slopes, $H_0\colon \beta_{12} = \beta_{13} = 0$: partial $F$-test with $m - 1 = 2$ numerator degrees of freedom.
    \end{itemize}
\end{frame}


\begin{frame}{Summary}
    \begin{itemize}
        \item An indicator variable encodes a binary category as $0/1$; a factor with $m$ levels needs $m-1$ indicators plus an intercept.
        \item Indicators shift the intercept: they give parallel lines with common slope. Coefficients are differences from the reference level.
        \item Including all $m$ indicators with an intercept makes $\bX' \bX$ singular (dummy variable trap).
        \item The reference level changes coefficients, not the fit; test a whole factor with the partial $F$-test.
        \item Interaction terms $x_1 x_2$ let the slope depend on the category (or on another regressor); test them first with the extra-sum-of-squares $F$-test (RM versus FM).
        \item Tool life data: type matters ($F_0 = 121.8$); the slope difference is not significant ($p = 0.196$).
    \end{itemize}
\end{frame}
