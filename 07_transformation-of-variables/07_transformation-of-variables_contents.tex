\section{Model Inadequacies and Transformations}


\begin{frame}{Correcting Model Inadequacies}
\begin{itemize}
    \item Least-squares inference assumes a linear relationship, constant error variance, and normally distributed errors
    \item Residual plots reveal when these assumptions fail
    \item One standard remedy is to transform the data; weighting is a separate topic
    \item In this lecture: transforming $y$ to stabilize the variance, transforming $y$ or $x$ to linearize the model, and two analytical methods for choosing the power (Box--Cox for $y$, Box--Tidwell for $x$)
\end{itemize}
\end{frame}


\begin{frame}{Recognizing Nonconstant Variance}
\begin{itemize}
    \item Plot R-student against the fitted values $\hat{y}_i$
    \item Outward-opening funnel: the spread grows with $\E[y_i]$, common when $y$ is a count or an amount
    % \item Double bow: the spread is largest in the middle and small at the extremes, typical when $y$ is a proportion
    \item With nonconstant variance the OLS estimators remain unbiased but are no longer minimum-variance, and the usual $t$ and $F$ tests are unreliable
\end{itemize}
\end{frame}


\begin{frame}{Idea of a Variance-Stabilizing Transformation}
\begin{itemize}
    \item Suppose $\Var(y_i)$ depends on the mean $\E[y_i]$
    \item Replace $y$ by $y^* = h(y)$, with $h$ chosen so that $\Var(y^*)$ is approximately constant
    \item Then fit the model on the transformed scale
\end{itemize}
\begin{align*}
    y_i^* = \beta_0 + \beta_1 x_i + \varepsilon_i, \qquad \Var(\varepsilon_i) \approx \sigma^2
\end{align*}
\begin{itemize}
    \item The appropriate $h$ depends on how $\Var(y)$ is related to $\E[y]$
\end{itemize}
\end{frame}



\section{Variance-Stabilizing Transformations}


\begin{frame}{Variance-Stabilizing Transformations}
\begin{center}
{\renewcommand{\arraystretch}{1.4}
\small
\begin{tabular}{ll}
\toprule
Relationship of $\sigma^2$ to $\E[y]$ & Transformation \\
\midrule
$\sigma^2 \propto \text{constant}$ & $y^* = y$ (no transformation) \\
$\sigma^2 \propto \E[y]$ & $y^* = \sqrt{y}$ (Poisson data) \\
% $\sigma^2 \propto \E[y]\,(1 - \E[y])$ & $y^* = \sin^{-1}\!\big(\sqrt{y}\,\big)$ (binomial proportions, $0 \le y_i \le 1$) \\
$\sigma^2 \propto (\E[y])^2$ & $y^* = \ln y$ \\
$\sigma^2 \propto (\E[y])^3$ & $y^* = y^{-1/2}$ \\
$\sigma^2 \propto (\E[y])^4$ & $y^* = y^{-1}$ \\
\bottomrule
\end{tabular}}
\end{center}
\end{frame}


\begin{frame}{Where the Table Comes From: The Delta Method}
Let $\mu = \E[y]$. A first-order Taylor expansion of $h$ about $\mu$ gives
\begin{align*}
    h(y) \approx h(\mu) + h'(\mu)\,(y - \mu)
    \quad\Longrightarrow\quad
    \Var\big(h(y)\big) \approx \big[h'(\mu)\big]^2 \Var(y)
\end{align*}
\begin{itemize}
    \item $\Var(h(y))$ is approximately constant when $h'(\mu) \propto 1/\sqrt{\Var(y)}$
    \item So $h$ is found by integrating one over the standard deviation, viewed as a function of the mean
\end{itemize}
\end{frame}


\begin{frame}{The Delta Method for Power Relationships}
Suppose the standard deviation is a power of the mean, $\sigma \propto \mu^{a}$, that is $\sigma^2 \propto \mu^{2a}$. Then
\begin{align*}
    h'(\mu) \propto \mu^{-a}
    \quad\Longrightarrow\quad
    h(y) =
    \begin{cases}
        y^{1-a}, & a \neq 1 \\
        \ln y, & a = 1
    \end{cases}
\end{align*}
\begin{itemize}
    \item $a = 1/2$: $y^{1/2}$; \ $a = 1$: $\ln y$; \ $a = 3/2$: $y^{-1/2}$; \ $a = 2$: $y^{-1}$, the rows of the table
    \item The faster $\sigma$ grows with $\mu$, the smaller the power $1 - a$, and the stronger the transformation
\end{itemize}
\end{frame}


\begin{frame}{Reading the Table}
\begin{itemize}
    \item The stronger the dependence of $\sigma^2$ on $\E[y]$, the stronger the transformation: $\sqrt{y}$, then $\ln y$, then $y^{-1/2}$, then $y^{-1}$
    \item Poisson counts have $\Var(y) = \E[y]$, so $\sigma^2 \propto \E[y]$ and the square root applies
    % \item A proportion $y = X/m$ with $X \sim \text{Binomial}(m, p)$ has $\Var(y) = p(1-p)/m$, so $\sigma^2 \propto \E[y](1 - \E[y])$ and the arcsine square root applies
    \item $\sigma^2 \propto (\E[y])^2$ means the standard deviation is proportional to the mean, which calls for the logarithm
\end{itemize}
\end{frame}


\begin{frame}{Choosing a Transformation in Practice}
\begin{enumerate}
    \item Fit the model by OLS on the original $y$ and plot R-student against $\hat{y}_i$
    \item If the pattern of spread matches a row of the table, apply that transformation
    \item Refit with $y^*$ and examine the residual plots again
    \item If the variance is still not constant, try a stronger or weaker transformation
\end{enumerate}
The exact relationship between $\sigma^2$ and $\E[y]$ is rarely known, so the choice is often made by trial and error.
\end{frame}


\begin{frame}{Consequences of Transforming the Response}
\begin{itemize}
    \item Estimates and tests refer to the $y^*$ scale; for example, coefficients describe the change in $\sqrt{y}$, not in $y$
    \item $R^2$ values from models with different responses cannot be compared; to compare fits on the $y$ scale, use for example the squared correlation between $y_i$ and the back-transformed fitted values $h^{-1}(\hat{y}_i^*)$
    \item Transforming $y$ also changes the shape of the relationship and the error distribution, so all residual checks must be repeated
\end{itemize}
\end{frame}


\begin{frame}{Back-Transforming to the Original Scale}
\begin{itemize}
    \item $h$ is monotone, so it preserves quantiles: a prediction interval $(L^*, U^*)$ for $y^*$ gives the prediction interval $\big(h^{-1}(L^*),\, h^{-1}(U^*)\big)$ for $y$ (endpoints swap if $h$ is decreasing, as for $y^{-1}$)
    \item The fitted value $h^{-1}(\hat{y}^*)$ estimates the \emph{median} of $y$, not its mean, when the errors on the $y^*$ scale are symmetric; likewise a confidence interval for $\E[y^*]$ back-transforms to one for the median of $y$
    \item Log scale, $y^* \sim N(\mu^*, \sigma^2)$: the median of $y$ is $e^{\mu^*}$ but $\E[y] = e^{\mu^* + \sigma^2/2}$
    \item Square-root scale: $y = (y^*)^2$, so $\E[y] = (\mu^*)^2 + \sigma^2$
\end{itemize}
\end{frame}

\section{Transformations to Linearize the Model}


\begin{frame}{Linearizable Nonlinear Functions}
\begin{center}
{\renewcommand{\arraystretch}{1.4}
\small
\begin{tabular}{lll}
\toprule
Function & Transformation & Linear form \\
\midrule
$y = \beta_0 x^{\beta_1}$ & $y^* = \ln y$, $x^* = \ln x$ & $y^* = \ln \beta_0 + \beta_1 x^*$ \\
$y = \beta_0 e^{\beta_1 x}$ & $y^* = \ln y$ & $y^* = \ln \beta_0 + \beta_1 x$ \\
$y = \beta_0 + \beta_1 \ln x$ & $x^* = \ln x$ & $y = \beta_0 + \beta_1 x^*$ \\
$y = \beta_0 + \dfrac{\beta_1}{x}$ & $x^* = \dfrac{1}{x}$ & $y = \beta_0 + \beta_1 x^*$ \\
$y = \dfrac{x}{\beta_0 x - \beta_1}$ & $y^* = \dfrac{1}{y}$, $x^* = \dfrac{1}{x}$ & $y^* = \beta_0 - \beta_1 x^*$ \\
\bottomrule
\end{tabular}}
\end{center}
\end{frame}

\begin{frame}{The Error Structure Matters}
\begin{itemize}
    \item Linearizing the exponential function assumes the error is multiplicative
\end{itemize}
\begin{align*}
    y = \beta_0 e^{\beta_1 x}\,\varepsilon \quad\Longrightarrow\quad \ln y = \ln \beta_0 + \beta_1 x + \ln \varepsilon
\end{align*}
\begin{itemize}
    \item The linearized model is appropriate if $\ln \varepsilon$ has mean zero, constant variance, and is normal
    \item If instead the error is additive, $y = \beta_0 e^{\beta_1 x} + \varepsilon$, taking logarithms does not give a linear model with additive error, and nonlinear regression is needed
\end{itemize}
\end{frame}


\begin{frame}{Interpreting Coefficients After a Log Transformation}
\begin{itemize}
    \item $\ln y = \beta_0 + \beta_1 x + \varepsilon$: a unit increase in $x$ multiplies the median of $y$ by $e^{\beta_1}$, a change of $100(e^{\beta_1} - 1)\%$, which is about $100\beta_1\%$ when $\beta_1$ is small
    \item $y = \beta_0 + \beta_1 \ln x + \varepsilon$: a $1\%$ increase in $x$ changes $\E[y]$ by $\beta_1 \ln 1.01 \approx \beta_1/100$
    \item $\ln y = \beta_0 + \beta_1 \ln x + \varepsilon$: a $1\%$ increase in $x$ multiplies the median of $y$ by $1.01^{\beta_1}$, about a $\beta_1\%$ change; $\beta_1$ is the elasticity of $y$ with respect to $x$
\end{itemize}
\end{frame}


\begin{frame}{Transform $x$ or Transform $y$?}
\begin{itemize}
    \item Transforming $x$ alone leaves the distribution of the errors unchanged
    \item Transforming $y$ changes both the shape of the relationship and the error distribution
    \item Curved scatter with roughly constant variance: transform $x$
    \item Curved scatter with nonconstant variance: transform $y$, which may fix both problems
\end{itemize}
\end{frame}

\section{The Box--Cox Method}


\begin{frame}{Motivation for the Box--Cox Method}
\begin{itemize}
    \item The variance-stabilizing table requires knowing how $\sigma^2$ depends on $\E[y]$, which is usually unknown
    \item Box and Cox (1964): estimate the power of the transformation from the data
    \item Considers the power family $y^\lambda$, with $\lambda$ estimated together with $\bbeta$ by maximum likelihood
    \item Aims to correct nonnormality and nonconstant variance
    \item Requires $y_i > 0$; if some $y_i \le 0$, shift to $y + c$ with $c$ large enough, or use the Yeo--Johnson family, which allows any real $y$
\end{itemize}
\end{frame}


\begin{frame}{The Power Transformation Family}
\begin{align*}
    y^{(\lambda)} =
    \begin{cases}
        \dfrac{y^\lambda - 1}{\lambda}, & \lambda \neq 0 \\[1ex]
        \ln y, & \lambda = 0
    \end{cases}
\end{align*}
\begin{itemize}
    \item As $\lambda \to 0$, $(y^\lambda - 1)/\lambda \to \ln y$, so the logarithm belongs to the family
    \item $\lambda = 1$: no transformation, $\lambda = 1/2$: square root, $\lambda = -1$: reciprocal
\end{itemize}
\end{frame}


\begin{frame}{Why $\SSRes$ Cannot Be Compared Across $\lambda$}
\begin{itemize}
    \item The units and size of $y^{(\lambda)}$ change with $\lambda$, so the raw $\SSRes$ changes with $\lambda$ even when the fit does not
    \item Electric utility data, unscaled $\SSRes$:
\end{itemize}
\begin{center}
{\renewcommand{\arraystretch}{1.2}
\small
\begin{tabular}{lrrr}
\toprule
Response & $y$ & $\sqrt{y}$ & $\ln y$ \\
\midrule
$\SSRes$ & $126.87$ & $10.98$ & $27.53$ \\
\bottomrule
\end{tabular}}
\end{center}
\begin{itemize}
    \item $\sqrt{y}$ is not ``ten times better'' than $y$: its values are simply smaller
\end{itemize}
\end{frame}


\begin{frame}{The Scaled Response}
\begin{itemize}
    \item Fix: divide by a factor involving the geometric mean $\dot{y}$, which puts every $\lambda$ on the units of $y$
\end{itemize}
\begin{align*}
    \tilde{y}^{(\lambda)} =
    \begin{cases}
        \dfrac{y^\lambda - 1}{\lambda\,\dot{y}^{\lambda - 1}}, & \lambda \neq 0 \\[1ex]
        \dot{y} \ln y, & \lambda = 0
    \end{cases}
    \qquad
    \dot{y} = \exp\!\left[\frac{1}{n}\sum_{i=1}^{n} \ln y_i\right]
\end{align*}
\begin{itemize}
    \item $\tilde{y}^{(\lambda)} = y^{(\lambda)}/\dot{y}^{\lambda - 1}$, so for each $\lambda$ the two versions differ by a constant factor and give the same fitted model up to scale
    \item Check: at $\lambda = 1$, $\tilde{y}^{(1)} = y - 1$, so its $\SSRes$ is that of the straight-line fit to $y$
\end{itemize}
\end{frame}


\begin{frame}{Model and Estimation}
\begin{align*}
    \tilde{\by}^{(\lambda)} = \bX\bbeta + \bvarepsilon, \qquad \bvarepsilon \sim N(\bzero, \sigma^2 \bI)
\end{align*}
\begin{itemize}
    \item For fixed $\lambda$, $\hat{\bbeta} = (\bX'\bX)^{-1}\bX'\tilde{\by}^{(\lambda)}$
    \item Let $\SSRes(\lambda)$ be the residual sum of squares from this fit
\end{itemize}
\begin{align*}
    \SSRes(\lambda) = \sum_{i=1}^{n} \big( \tilde{y}_i^{(\lambda)} - \hat{\tilde{y}}_i^{(\lambda)} \big)^2
\end{align*}
\begin{itemize}
    \item With the scaled response, the maximum likelihood estimate $\hat{\lambda}$ is the value that minimizes $\SSRes(\lambda)$ (derivation next)
\end{itemize}
\end{frame}


\begin{frame}{Why Minimizing $\SSRes(\lambda)$ Is Maximum Likelihood (1)}
\begin{itemize}
    \item Assume $y_i^{(\lambda)} = \bx_i'\bbeta + \varepsilon_i$ with $\varepsilon_i \sim N(0, \sigma^2)$ independent (unscaled family)
    \item The density of $y_i$ is the normal density of $y_i^{(\lambda)}$ times the Jacobian $\big|\mathrm{d} y_i^{(\lambda)}/\mathrm{d} y_i\big| = y_i^{\lambda - 1}$
\end{itemize}
\begin{align*}
    \ell(\bbeta, \sigma^2, \lambda) = -\frac{n}{2}\ln(2\pi\sigma^2) - \frac{1}{2\sigma^2}\sum_{i=1}^{n}\big(y_i^{(\lambda)} - \bx_i'\bbeta\big)^2 + (\lambda - 1)\sum_{i=1}^{n} \ln y_i
\end{align*}
\begin{itemize}
    \item Without the Jacobian term, the likelihood would compare densities of different variables $y^{(\lambda)}$ rather than of the observed $y$
\end{itemize}
\end{frame}


\begin{frame}{Why Minimizing $\SSRes(\lambda)$ Is Maximum Likelihood (2)}
\begin{itemize}
    \item For fixed $\lambda$, maximize over $\bbeta$ and $\sigma^2$: OLS gives $\hat{\bbeta}$, and $\hat{\sigma}^2 = S(\lambda)/n$, where $S(\lambda)$ is the unscaled residual sum of squares
    \item Substituting gives the profile log-likelihood
\end{itemize}
\begin{align*}
    \ell_p(\lambda) = \text{const} - \frac{n}{2}\ln S(\lambda) + (\lambda - 1)\sum_{i=1}^{n} \ln y_i
\end{align*}
\begin{itemize}
    \item Since $\sum_i \ln y_i = n \ln \dot{y}$, the last term is $\tfrac{n}{2}\ln \dot{y}^{\,2(\lambda - 1)}$, so
\end{itemize}
\begin{align*}
    \ell_p(\lambda) = \text{const} - \frac{n}{2}\ln \frac{S(\lambda)}{\dot{y}^{\,2(\lambda - 1)}} = \text{const} - \frac{n}{2}\ln \SSRes(\lambda)
\end{align*}
\begin{itemize}
    \item Dividing $y^{(\lambda)}$ by $\dot{y}^{\lambda - 1}$ divides $S(\lambda)$ by $\dot{y}^{2(\lambda-1)}$: the scaling absorbs the Jacobian
\end{itemize}
\end{frame}


\begin{frame}{The Box--Cox Procedure}
\begin{enumerate}
    \item Choose a grid of values of $\lambda$, for example from $-2$ to $2$
    \item For each $\lambda$, compute the scaled $\tilde{y}^{(\lambda)}$, fit the model, and record $\SSRes(\lambda)$
    \item Plot $\SSRes(\lambda)$ against $\lambda$ and locate the minimum $\hat{\lambda}$
    \item Analyze the simple power $y^\lambda$ (not the scaled version), using a convenient value of $\lambda$ near $\hat{\lambda}$, and recheck the residuals
\end{enumerate}
Later inference treats $\lambda$ as known, so it is only approximate.
\end{frame}


\begin{frame}{Confidence Interval for $\lambda$}
\begin{align*}
    \SSstar = \SSRes(\hat{\lambda})\left(1 + \frac{t_{\alpha/2,\,\nu}^2}{\nu}\right)
\end{align*}
\begin{itemize}
    \item $\nu$ is the residual degrees of freedom
    \item A $100(1-\alpha)\%$ confidence interval for $\lambda$ contains all $\lambda$ with $\SSRes(\lambda) \le \SSstar$
    \item If the interval contains $1$, the data give no evidence that a transformation is needed
    \item Otherwise choose a convenient value inside the interval, such as $-1$, $0$, $1/2$
\end{itemize}
\end{frame}


\begin{frame}{Log-Likelihood Form and \texttt{MASS::boxcox}}
\begin{itemize}
    \item Up to a constant, the profile log-likelihood of $\lambda$ is (as derived above)
\end{itemize}
\begin{align*}
    L(\lambda) = -\frac{n}{2} \ln \SSRes(\lambda)
\end{align*}
\begin{itemize}
    \item The likelihood-based interval keeps all $\lambda$ with $L(\hat{\lambda}) - L(\lambda) \le \tfrac{1}{2}\chi^2_{\alpha,\,1}$
    \item Equivalently, $\SSRes(\lambda) \le \SSRes(\hat{\lambda})\exp\!\big(\chi^2_{\alpha,\,1}/n\big)$, which agrees closely with $\SSstar$ unless $n$ is small
    \item \texttt{MASS::boxcox} plots $L(\lambda)$ with this 95\% interval
    \item \texttt{car::powerTransform} reports $\hat{\lambda}$, a Wald interval, and likelihood-ratio tests of $\lambda = 1$ and $\lambda = 0$
\end{itemize}
\end{frame}

\section{The Box--Tidwell Method}


\begin{frame}{Motivation for the Box--Tidwell Method}
\begin{itemize}
    \item Box--Cox transforms $y$; when the problem is nonlinearity with constant variance, transform the regressor instead
    \item Model, with unknown power $\alpha$:
\end{itemize}
\begin{align*}
    y = \beta_0 + \beta_1 \xi + \varepsilon, \qquad
    \xi =
    \begin{cases}
        x^\alpha, & \alpha \neq 0 \\
        \ln x, & \alpha = 0
    \end{cases}
\end{align*}
\begin{itemize}
    \item Box and Tidwell (1962) estimate $\alpha$ iteratively
    \item Special case: $\alpha = 0$: logarithm
\end{itemize}
\end{frame}


\begin{frame}{Linearizing the Model in $\alpha$: Setup}
\begin{itemize}
    \item The model $\E[y] = \beta_0 + \beta_1 x^\alpha$ is nonlinear in the unknown power $\alpha$
    \item Idea: replace it by a first-order Taylor series in $\alpha$ about a starting value $\alpha_0$, treating $x$, $\beta_0$, $\beta_1$ as fixed
\end{itemize}
\begin{align*}
    f(\alpha) \approx f(\alpha_0) + (\alpha - \alpha_0)\, f'(\alpha_0), \qquad f(\alpha) = \beta_0 + \beta_1 x^\alpha
\end{align*}
\begin{itemize}
    \item Take $\alpha_0 = 1$: then $x^{\alpha_0} = x$, the untransformed straight-line model
    \item We need $f'(\alpha)$, which requires $\partial x^\alpha / \partial \alpha$
\end{itemize}
\end{frame}


\begin{frame}{Step 1: Differentiate $x^\alpha$ with Respect to $\alpha$}
\begin{itemize}
    \item Here $x$ is a constant and $\alpha$ is the variable, so $x^\alpha$ is an exponential function of $\alpha$, not a power function of $x$
    \item Rewrite it with base $e$ (this requires $x > 0$)
\end{itemize}
\begin{align*}
    x^\alpha = e^{\alpha \ln x}
\end{align*}
\begin{itemize}
    \item $\ln x$ is a constant with respect to $\alpha$, so the chain rule gives
\end{itemize}
\begin{align*}
    \frac{\partial}{\partial \alpha}\, x^\alpha = \frac{\partial}{\partial \alpha}\, e^{\alpha \ln x} = (\ln x)\, e^{\alpha \ln x} = x^\alpha \ln x
\end{align*}
\begin{itemize}
    \item Do not confuse this with $\partial x^\alpha / \partial x = \alpha x^{\alpha - 1}$, which differentiates with respect to $x$
\end{itemize}
\end{frame}


\begin{frame}{Step 2: Evaluate the Taylor Expansion at $\alpha_0 = 1$}
$\beta_0$ and $\beta_1$ do not depend on $\alpha$, so differentiating $f(\alpha) = \beta_0 + \beta_1 x^\alpha$ gives
\begin{align*}
    f'(\alpha) = \beta_1 \frac{\partial x^\alpha}{\partial \alpha} = \beta_1\, x^\alpha \ln x
\end{align*}
Evaluate $f$ and $f'$ at $\alpha_0 = 1$:
\begin{align*}
    f(1) = \beta_0 + \beta_1 x, \qquad f'(1) = \beta_1\, x \ln x
\end{align*}
Substitute into $f(\alpha) \approx f(1) + (\alpha - 1)\, f'(1)$:
\begin{align*}
    \E[y] \approx \beta_0 + \beta_1 x + (\alpha - 1)\,\beta_1\, x \ln x
\end{align*}
\end{frame}


\begin{frame}{Step 3: Reparameterize to a Linear Model}
\begin{itemize}
    \item $(\alpha - 1)\beta_1$ is a single unknown constant, and $x \ln x$ can be computed from the data
\end{itemize}
\begin{align*}
    \gamma = (\alpha - 1)\beta_1, \qquad w = x \ln x
\end{align*}
\begin{align*}
    \E[y] \approx \beta_0 + \beta_1 x + \gamma w
\end{align*}
\begin{itemize}
    \item The model is linear in the regressors $x$ and $w$, so the coefficients can be estimated by OLS
    \item Recover the power by solving $\gamma = (\alpha - 1)\beta_1$ for $\alpha$: \ $\alpha = \gamma/\beta_1 + 1$
    \item $\gamma = 0$ exactly when $\alpha = 1$, that is, when no transformation of $x$ is needed
\end{itemize}
\end{frame}


\begin{frame}{One Iteration of the Procedure}
\begin{enumerate}
    \item Fit $y = \beta_0 + \beta_1 x + \varepsilon$ and record $\hat{\beta}_1$
    \item Fit $y = \beta_0^* + \beta_1^* x + \gamma w + \varepsilon$ with $w = x \ln x$ and record $\hat{\gamma}$
    \item Compute the first estimate of the power
\end{enumerate}
\begin{align*}
    \hat{\alpha}_1 = \frac{\hat{\gamma}}{\hat{\beta}_1} + 1
\end{align*}
\begin{itemize}
    \item The coefficients in step 2 are starred because the linearized model holds only approximately, so $\hat{\beta}_0^*$, $\hat{\beta}_1^*$ need not estimate $\beta_0$, $\beta_1$ well
    \item Following Box and Tidwell, $\hat{\beta}_1$ is taken from the fit in step 1, not from step 2
\end{itemize}
\end{frame}


\begin{frame}{Iterating}
\begin{itemize}
    \item Set $x^{(1)} = x^{\hat{\alpha}_1}$ and repeat the procedure with $x^{(1)}$ in place of $x$
    \item Stop when $\hat{\alpha}_k \approx 1$, that is, when no further transformation is needed
    \item The overall estimate is the product $\hat{\alpha} = \prod_k \hat{\alpha}_k$, since a power of a power multiplies the exponents
    \item The $t$ statistic for $\hat{\gamma}$ gives an approximate test of $H_0\colon \alpha = 1$
    \item In R, \texttt{car::boxTidwell} runs the iterations and reports $\hat{\alpha}$ with this test
\end{itemize}
\end{frame}

\section{Summary}


\begin{frame}{Summary}
\begin{itemize}
    \item Nonconstant variance: transform $y$ using the variance-stabilizing table, then recheck the residuals
    \item Unknown variance--mean relationship: estimate the power of $y$ with the Box--Cox method
    \item Curved relationship with constant variance: transform $x$, guided by the shape of the scatter plot
    \item Unknown power for $x$: estimate it with the Box--Tidwell method
    \item After any transformation, repeat the residual analysis and interpret results on the transformed scale
    \item Back-transformed predictions estimate the median of $y$; log-scale coefficients are percentage changes
\end{itemize}
\end{frame}
