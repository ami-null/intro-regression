\section{Estimating the SLR Parameters}


\begin{frame}{Introduction}
	\begin{itemize}
		\item OLS: choose $\beta_0,\beta_1$ to minimize $\sum_i(y_i-\beta_0-\beta_1 x_i)^2$
		\item MLE: given a full distributional assumption on the errors, choose parameter values that make the observed data most probable
	\end{itemize}
\end{frame}


\begin{frame}{The Likelihood Function}
	The likelihood function is the joint density of the observed data, viewed as a function of the parameters rather than the data.

	\svspace{0.5em}

	Since $Y_i \sim N(\beta_0+\beta_1 x_i,\ \sigma^2)$, and due to the observations being independent, the likelihood function is:
	\begin{align}
		L(\beta_0,\beta_1,\sigma^2;\ \by,\bx) &= \prod_{i=1}^n \frac{1}{\sqrt{2\pi\sigma^2}}\exp\!\left(-\frac{(y_i-\beta_0-\beta_1 x_i)^2}{2\sigma^2}\right) \notag
	\end{align}

	\pause
	
	Since $\log$ is monotonically increasing, maximizing $L$ is equivalent to maximizing the log-likelihood:
	\begin{align}
		\ell(\beta_0,\beta_1,\sigma^2;\ \by,\bx) &= -\frac n2\ln(2\pi) - \frac n2\ln(\sigma^2) - \frac{1}{2\sigma^2}\sum_i(y_i-\beta_0-\beta_1 x_i)^2 \label{eq:loglik}
	\end{align}
\end{frame}


\begin{frame}{Maximizing Over $\beta_0$ and $\beta_1$}
	\begin{itemize}
		\item For fixed $\sigma^2$, only the last term of $\ell$ in Equation~\eqref{eq:loglik} depends on $\beta_0,\beta_1$
		\item Maximizing $\ell$ over $\beta_0,\beta_1$ $\iff$ minimizing $\sum_i(y_i-\beta_0-\beta_1 x_i)^2$, which is the same as the OLS criterion
	\end{itemize}
	\svspace{1em}
	\begin{itemize}
		\item[] \textbf{Question:} what does this imply about $\hat\beta_{0,MLE},\hat\beta_{1,MLE}$ versus the OLS estimates?
	\end{itemize}
	\svspace{1em}
	\begin{itemize}
		\item They are identical --- under the normal-error model, the the maximum likelihood estimates of \(\beta_0\) and \(\beta_1\) coincide exactly with the least-squares estimates
	\end{itemize}
\end{frame}


\section{Estimating the Error Variance}


\begin{frame}{MLE of $\sigma^2$}
	Differentiating $\ell$ (Equation~\eqref{eq:loglik}) with respect to $\sigma^2$ and solving:
	\begin{align}
		\frac{\partial \ell}{\partial \sigma^2} &= -\frac{n}{2\sigma^2} + \frac{SSE}{2\sigma^4} = 0 \notag\\
		\hat\sigma^2_{MLE} &= \frac{SSE}{n} \label{eq:mle-sigma2}
	\end{align}
	\begin{itemize}
		\item Compare to the OLS/unbiased estimator: $MSE = SSE/(n-2)$
		\item $\hat\sigma^2_{MLE}$ divides by $n$, not $n-2$ --- it is \emph{biased}: $E[\hat\sigma^2_{MLE}] = \frac{n-2}{n}\sigma^2$
		\item $MSE$ is the bias-corrected version of $\hat\sigma^2_{MLE}$
	\end{itemize}
\end{frame}


\begin{frame}{The Difference, in Our Example}
	Rocket-propellant data: $n=20$, $SSE=166{,}254.65$
	\begin{align*}
		\hat\sigma^2_{MLE} &= \frac{SSE}{n} = \frac{166{,}254.65}{20} = 8312.73 &&\text{by Equation~\eqref{eq:mle-sigma2}}\\
		MSE &= \frac{SSE}{n-2} = \frac{166{,}254.65}{18} = 9236.37
	\end{align*}
	\begin{itemize}
		\item The MLE underestimates $\sigma^2$ by about 10\% here --- $n=20$ is small enough for the bias to matter
		\item The two converge as $n$ grows
	\end{itemize}
\end{frame}


\begin{frame}{Where MLE and OLS Diverge, in General}
	\begin{center}
		\mode<presentation>{\small}
		\centering
		\begin{tabular}{lll}
			\toprule
			& OLS & MLE\\
			\midrule
			Needs a distribution? & No (mean, variance only) & Yes (full distribution)\\
			$\beta_0,\beta_1$ estimates & Least squares & Identical, under normality\\
			$\sigma^2$ estimate & Unbiased, $SSE/(n-2)$ & Biased, $SSE/n$\\
			Framework & Geometric & Probabilistic\\
			Optimality & Gauss--Markov (BLUE) & Asymptotic efficiency\\
			\bottomrule
		\end{tabular}
	\end{center}
\end{frame}


\section{Comparison}


\begin{frame}{Statistical Advantages of MLE}
	\begin{itemize}
		\item Asymptotically efficient: attains the Cram\'er--Rao lower bound as $n\to\infty$, under correct specification
		\item Provides a general toolbox: standard errors from the Fisher information, likelihood-ratio/Wald/score tests, AIC/BIC for model comparison
		\item Generalizes to models for which OLS has no natural analogue, for example: logistic regression, Poisson regression, and other generalized linear models (GLM)
		\item Consistent and asymptotically normal under mild regularity conditions
	\end{itemize}
\end{frame}


\begin{frame}{Where OLS Still Wins}
	\begin{itemize}
		\item Gauss--Markov: OLS is BLUE assuming only zero-mean, constant-variance, uncorrelated errors --- \emph{no} distributional assumption required
		\item MLE's efficiency and consistency rely on the assumed distribution being (at least approximately) correct
		\item Here, $\hat\beta_0,\hat\beta_1$ are unbiased in both approaches, but $\hat\sigma^2_{MLE}$ is biased while $\hat\sigma^2_{OLS} = MSE$ is not
	\end{itemize}
\end{frame}
