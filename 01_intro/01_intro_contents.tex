\section{Introduction to Regression}

\begin{frame}{What Is Regression?}
	Regression is a statistical technique for modeling the relationship between a \textbf{response} (dependent) variable and one or more \textbf{predictor} (independent) variables.
	\svspace{0.5em}

	Instead of treating events in a system as isolated, regression lets us ask: \emph{if this variable changes, how does that one respond, on average?}
	\svspace{0.5em}

	\begin{itemize}
		\item Response variable: what we want to explain or predict (denoted $y$)
		\item Predictor variable(s): what we use to explain it (denoted $x$)
		\item The regression equation summarizes the average relationship between them
	\end{itemize}
\end{frame}


\begin{frame}{Why Regression?}
	\begin{itemize}
		\item \textbf{Business:} does advertising spend increase sales, and by how much?
		\item \textbf{Medicine:} how does drug dosage affect blood pressure reduction?
		\item \textbf{Engineering:} how does curing temperature affect concrete strength?
		\item \textbf{Economics:} how does income relate to household spending?
		\item \textbf{Environment:} how does fertilizer amount relate to crop yield?
	\end{itemize}
\end{frame}


\section{Simple Linear Regression (SLR)}


\begin{frame}{The SLR Model Equation}
	For an outcome variable \(Y\) and a single predictor variable \(X\), the model is given as
	\[
		Y = \beta_0 + \beta_1 X + \epsilon, \qquad \epsilon \sim N(0, \sigma^2)
	\]

	\begin{itemize}
		\item $\beta_0$: the intercept --- expected or average value of \(Y\) when the value of \(X\) is zero
		\item $\beta_1$: the slope --- the expected or average change in \(Y\) for a one-unit increase \(X\)
		\item $\epsilon_i$: random error --- everything affecting \(Y\) that \(X\) does not capture
		\item $\beta_0, \beta_1$ are unknown \textbf{population parameters}; we estimate them from data as $\hat\beta_0, \hat\beta_1$
	\end{itemize}
\end{frame}


\begin{frame}{The SLR Model in Terms of Observed Values}
	For a single predictor $X$ (advertising spend) and response $Y$ (sales), and observed data points \((x_i, y_i)\), \(i = 1, 2,...,n\), the SLR model is
	\[
	y_i = \beta_0 + \beta_1 x_i + \varepsilon_i, \qquad \varepsilon_i \overset{iid}{\sim} N(0, \sigma^2), \qquad i = 1, \dots, n.
	\]

	The fitted line is given by
	\[
		\hat{y}_i = \hat{\beta}_0 + \hat{\beta}_1 x_i.
	\]
	
	What are the interpretations of \(\hat{\beta}_0\) and \(\hat{\beta}_1\) in this context?

	% \begin{itemize}
	% 	\item $\beta_0$: the intercept --- expected sales when advertising spend is zero
	% 	\item $\beta_1$: the slope --- the expected change in sales for a one-unit increase in advertising spend
	% 	\item $\varepsilon_i$: random error --- everything affecting sales that $x$ does not capture
	% 	\item $\beta_0, \beta_1$ are unknown \textbf{population parameters}; we estimate them from data as $\hat\beta_0, \hat\beta_1$
	% \end{itemize}

\end{frame}

\begin{frame}{Assumptions of the SLR Model}
	\begin{itemize}
		\item \textbf{Linearity:} the mean of $y$ is a linear function of $x$, i.e.\ $E[\varepsilon_i] = 0$
		\item \textbf{Independence:} $\operatorname{Cov}(\varepsilon_i, \varepsilon_j) = 0$ for $i \neq j$
		\item \textbf{Normality}: $\varepsilon_i \sim N(0, \sigma^2)$
		\item \textbf{Constant variance (homoscedasticity):} $\operatorname{Var}(\varepsilon_i) = \sigma^2$ for all $i$
		\item $x_i$ values are fixed, known constants, measured without error
	\end{itemize}
\end{frame}

\begin{frame}{Residuals}

			\]
			Ordinary Least Squares (OLS) chooses $\hat\beta_0, \hat\beta_1$ to make these residuals collectively as small as possible.

\begin{frame}{Setting Up the Least Squares Problem}
	OLS minimizes the sum of squared residuals:
	\[
		S(\beta_0, \beta_1) = \sum_{i=1}^n \varepsilon_i^2 = \sum_{i=1}^n (y_i - \beta_0 - \beta_1 x_i)^2
	\]

	\pause
	\svspace{0.5em}
	To minimize $S(\beta_0,\beta_1)$, we take partial derivatives with respect to $\beta_0$ and $\beta_1$, set them equal to zero, and solve.
	
	\pause
	\svspace{0.5em}
	Since $S$ is a quadratic (convex) function, this stationary point is the global minimum.
\end{frame}


\begin{frame}{First-Order Conditions}
	Differentiate $S(\beta_0,\beta_1) = \sum_i (y_i - \beta_0 - \beta_1 x_i)^2$:
	\pause
	\[
		\frac{\partial S}{\partial \beta_0} = -2\sum_{i=1}^n (y_i - \beta_0 - \beta_1 x_i)
	\]
	\pause
	\[
		\frac{\partial S}{\partial \beta_1} = -2\sum_{i=1}^n x_i(y_i - \beta_0 - \beta_1 x_i)
	\]
	\pause
	Setting both equal to zero and evaluating at the minimizing values $\hat\beta_0, \hat\beta_1$ gives the \textbf{normal equations}.
\end{frame}


\begin{frame}{The Normal Equations}
	\[
		\sum_{i=1}^n y_i = n\hat\beta_0 + \hat\beta_1 \sum_{i=1}^n x_i
	\]
	\[
		\sum_{i=1}^n x_i y_i = \hat\beta_0 \sum_{i=1}^n x_i + \hat\beta_1 \sum_{i=1}^n x_i^2
	\]
	\svspace{0.5em}
	\pause
	This is a system of two linear equations in two unknowns, $\hat\beta_0$ and $\hat\beta_1$. We solve it directly using algebra.
\end{frame}


\begin{frame}{Solving for $\hat\beta_1$}
	From the first normal equation, dividing by $n$:
	\[
		\hat\beta_0 = \bar y - \hat\beta_1 \bar x
	\]
	\pause
	Substitute into the second normal equation:
	\[
		\sum_i x_i y_i = (\bar y - \hat\beta_1 \bar x)\sum_i x_i + \hat\beta_1 \sum_i x_i^2
	\]
	\pause
	Since $\sum_i x_i = n\bar x$, this becomes

	\begin{align*}
		\sum_i x_i y_i - n\bar x \bar y &= \hat\beta_1\left(\sum_i x_i^2 - n\bar x^2\right) \\
		\hat\beta_1 &= \frac{\sum_i x_i y_i - n\bar x \bar y}{\left(\sum_i x_i^2 - n\bar x^2\right)}
	\end{align*}
\end{frame}


\begin{frame}{Solving for $\hat\beta_1$ (cont.)}
	Define
	\[
		S_{xy} = \sum_i (x_i - \bar x)(y_i - \bar y) = \sum_i x_i y_i - n\bar x\bar y, \qquad
		S_{xx} = \sum_i (x_i - \bar x)^2 = \sum_i x_i^2 - n\bar x^2
	\]
	\pause
	\[
		\boxed{\hat\beta_1 = \frac{S_{xy}}{S_{xx}}}
	\]
\end{frame}


\begin{frame}{Solving for $\hat\beta_0$}
	Having solved for $\hat\beta_1$, we plug back into the first normal equation:
	\[
		\boxed{\hat\beta_0 = \bar y - \hat\beta_1 \bar x}
	\]
	\svspace{0.5em}
	\pause
	Interpretation for our example: $\hat\beta_1$ is the estimated extra sales revenue per extra unit of advertising spend, and the fitted line passes through the point of averages $(\bar x, \bar y)$.
\end{frame}


\begin{frame}{OLS Estimators --- Summary}
	\[
		\hat\beta_1 = \frac{S_{xy}}{S_{xx}}, \qquad \hat\beta_0 = \bar y - \hat\beta_1 \bar x, \qquad \hat y_i = \hat\beta_0 + \hat\beta_1 x_i
	\]
\end{frame}


\begin{frame}{Estimator of \texorpdfstring{\(\beta_1\)}{beta1} in terms of Covariance}
	Recall that
	\[
		\operatorname{CoV}(X, Y) = \frac{\sum_i (x_i - \bar x)(y_i - \bar y)}{n-1} = \frac{S_{xy}}{n-1} 
	\]
	and
	\[
		\operatorname{V}(X) = \frac{\sum_i (x_i - \bar x)^2}{n-1} = \frac{S_{xx}}{n-1},
	\]
	hence,
	\[
		\hat{\beta}_1 = \frac{\operatorname{CoV}(X, Y)}{\operatorname{V}(X)}.
	\]
\end{frame}


\begin{frame}{Relationship between \texorpdfstring{\(\hat{\beta}_1\)}{beta1hat} and the Correlation Coefficient}
	Recall that the population correlation coefficient \(\rho\) is estimated by
	\[
		r = \frac{\operatorname{Cov}(X, Y)}{\sqrt{\operatorname{V}(X) \cdot \operatorname{V}(Y)}},
	\]
	therefore
	\[
		r = \frac{\sqrt{\operatorname{V}(X)}}{\sqrt{\operatorname{V}(Y)}} \hat{\beta}_1.
	\]
\end{frame}

\section{Properties of the OLS Estimators}


\begin{frame}{Rewriting $\hat\beta_1$ as a Linear Combination of the $y_i$}
	Since $\sum_i (x_i - \bar x)\bar y = 0$, we can write
	\[
		\hat\beta_1 = \frac{\sum_i (x_i - \bar x)(y_i - \bar y)}{S_{xx}} = \frac{\sum_i (x_i - \bar x) y_i}{S_{xx}} = \sum_{i=1}^n c_i y_i, \qquad c_i = \frac{x_i - \bar x}{S_{xx}}
	\]
	\pause
	The weights $c_i$ satisfy two useful properties, both provable directly from the definition of $S_{xx}$:
	\[
		\sum_i c_i = 0, \qquad \sum_i c_i x_i = 1, \qquad \sum_i c_i^2 = \frac{1}{S_{xx}}
	\]
	These properties are needed to establish the unbiasedness of the estimators and find their variances.
\end{frame}


\begin{frame}{Unbiasedness of $\hat\beta_1$}
	Treating the $x_i$ as fixed and using $y_i = \beta_0 + \beta_1 x_i + \varepsilon_i$:
	\[
		E[\hat\beta_1] = E\left[\sum_i c_i y_i\right] = \sum_i c_i E[y_i] = \sum_i c_i(\beta_0 + \beta_1 x_i)
	\]
	\pause
	\[
		= \beta_0 \underbrace{\sum_i c_i}_{=0} + \beta_1 \underbrace{\sum_i c_i x_i}_{=1} = \beta_1
	\]
	\pause
	\[
		\boxed{E[\hat\beta_1] = \beta_1}
	\]
	On average, across repeated samples, the OLS slope estimate equals the true parameter.
\end{frame}


\begin{frame}{A Preliminary Result: $E[\bar y]$}
	Since $y_i = \beta_0 + \beta_1 x_i + \varepsilon_i$ with $E[\varepsilon_i] = 0$:
	\[
		E[y_i] = \beta_0 + \beta_1 x_i
	\]
	\pause
	Averaging over $i = 1, \dots, n$:
	\[
		E[\bar y] = \frac{1}{n}\sum_i E[y_i] = \frac{1}{n}\left(n\beta_0 + \beta_1\sum_i x_i\right) = \beta_0 + \beta_1\bar x
	\]
\end{frame}


\begin{frame}{Unbiasedness of $\hat\beta_0$}
	Using $\hat\beta_0 = \bar y - \hat\beta_1 \bar x$ and $E[\bar y] = \beta_0 + \beta_1 \bar x$:
	\[
		E[\hat\beta_0] = E[\bar y] - \bar x\, E[\hat\beta_1] = (\beta_0 + \beta_1 \bar x) - \bar x \beta_1
	\]
	\pause
	\[
		\boxed{E[\hat\beta_0] = \beta_0}
	\]
	Both OLS estimators are unbiased.
\end{frame}


\begin{frame}{Variance of the OLS Estimators}
	Because the $\varepsilon_i$ are independent with common variance $\sigma^2$, and $\hat\beta_1 = \sum_i c_i y_i$ with $\operatorname{Var}(y_i) = \sigma^2$:
	\[
		\operatorname{Var}(\hat\beta_1) = \sum_i c_i^2 \operatorname{Var}(y_i) = \sigma^2 \sum_i c_i^2 = \frac{\sigma^2}{S_{xx}}
	\]
	\pause
	A similar calculation for $\hat\beta_0 = \bar y - \hat\beta_1 \bar x$ gives
	\[
		\operatorname{Var}(\hat\beta_0) = \sigma^2\left(\frac{1}{n} + \frac{\bar x^2}{S_{xx}}\right)
	\]
	\pause
	More spread-out $x$ values (larger $S_{xx}$) give more precise slope estimates --- intuitively, advertising spend that varies a lot from month to month tells us more about its effect on sales than spend that barely changes.
\end{frame}


\begin{frame}{Variance of $\hat\beta_0$}
	Using $\hat\beta_0 = \bar y - \hat\beta_1\bar x$:
	\begin{align*}
		\operatorname{Var}(\hat\beta_0) &= \operatorname{Var}(\bar y) + \bar x^2\operatorname{Var}(\hat\beta_1) - 2\operatorname{Cov}(\bar y, \hat\beta_1 \bar x) \\
		&= \frac{\sigma^2}{n} + \bar x^2\cdot\frac{\sigma^2}{S_{xx}} - 2\bar x\operatorname{Cov}\left(\bar y, \sum_{i=1}^n c_i y_i\right) \\
		&= \sigma^2\left(\frac{1}{n} + \frac{\bar x^2}{S_{xx}}\right) - 2\bar x\sum_{i=1}^n c_i\operatorname{Cov}(\bar y, y_i)
	\end{align*}
	
\end{frame}


\begin{frame}
	\begin{align*}
		&= \sigma^2\left(\frac{1}{n} + \frac{\bar x^2}{S_{xx}}\right) - 2\bar x\sum_{i=1}^n c_i\operatorname{Cov}\left(\frac{1}{n}\sum_{j=1}^n y_j,\ y_i\right) \\
		&= \sigma^2\left(\frac{1}{n} + \frac{\bar x^2}{S_{xx}}\right) - 2\bar x\sum_{i=1}^n c_i\cdot\frac{\sigma^2}{n} \\
		&= \sigma^2\left(\frac{1}{n} + \frac{\bar x^2}{S_{xx}}\right) - 2\bar x\cdot 0\cdot\sigma^2
	\end{align*}

	\pause
	\[
		\boxed{\operatorname{Var}(\hat\beta_0) = \sigma^2\left(\frac{1}{n} + \frac{\bar x^2}{S_{xx}}\right)}
	\]
\end{frame}


\begin{frame}{The Gauss--Markov Theorem: OLS Is BLUE}
	Among all estimators of $\beta_0$ and $\beta_1$ that are:
	\begin{itemize}
		\item \textbf{L}inear in the $y_i$, and
		\item \textbf{U}nbiased,
	\end{itemize}
	the OLS estimators $\hat\beta_0, \hat\beta_1$ have the smallest possible variance.
	\svspace{0.5em}
	\pause
	This is why OLS is called \textbf{BLUE}: the \textbf{B}est \textbf{L}inear \textbf{U}nbiased \textbf{E}stimator. ``Best'' here means minimum variance, not that OLS makes no errors --- it means no other linear, unbiased method can do better, under the SLR assumptions.
\end{frame}


\section{Other Useful Properties of the Least-Squares Fit}


\begin{frame}{Other Useful Properties of the Least-Squares Fit}
	Beyond unbiasedness and BLUE, the OLS fit has several algebraic properties that follow directly from the normal equations --- worth knowing since they show up in diagnostics and in software output.
	\svspace{0.5em}
	\begin{itemize}
		\item The residuals always sum to zero: $\sum_i \hat\varepsilon_i = 0$
		\item The observed and fitted values have the same sum: $\sum_i y_i = \sum_i \hat y_i$
		\item The fitted line always passes through the centroid $(\bar x, \bar y)$
		\item Residuals weighted by $x_i$ sum to zero: $\sum_i x_i \hat\varepsilon_i = 0$
		\item Residuals weighted by $\hat y_i$ sum to zero: $\sum_i \hat y_i \hat\varepsilon_i = 0$
	\end{itemize}
\end{frame}


\begin{frame}{Property 1 \& 2: Residuals Sum to Zero}
	Recall the first normal equation from earlier:
	\[
		\sum_i y_i = n\hat\beta_0 + \hat\beta_1\sum_i x_i
	\]
	\pause
	Since $\hat y_i = \hat\beta_0 + \hat\beta_1 x_i$, summing over $i$ gives
	\[
		\sum_i \hat y_i = n\hat\beta_0 + \hat\beta_1\sum_i x_i
	\]
	\pause
	The right-hand sides are identical, so
	\[
		\boxed{\sum_i y_i = \sum_i \hat y_i} \quad\Longrightarrow\quad \sum_i(y_i - \hat y_i) = \sum_i \hat\varepsilon_i = 0
	\]
	This is not a coincidence --- it is forced by the very equation OLS solves to find $\hat\beta_0$.
\end{frame}


\begin{frame}{Property 3: The Fitted Line Passes Through $(\bar x, \bar y)$}
	Divide the previous result $\sum_i y_i = \sum_i \hat y_i$ by $n$:
	\[
		\bar y = \overline{\hat y} = \frac{1}{n}\sum_i(\hat\beta_0 + \hat\beta_1 x_i) = \hat\beta_0 + \hat\beta_1 \bar x
	\]
	\pause
	So the point $(\bar x, \bar y)$ satisfies the fitted equation exactly --- the \textbf{centroid} of the data always lies on the regression line.
	\svspace{0.5em}
	This matches how we solved for $\hat\beta_0$ earlier: we built the line to pass through $(\bar x,\bar y)$ by construction.
\end{frame}


\begin{frame}{Property 4 \& 5: Weighted Residuals Sum to Zero}
	Recall the second normal equation:
	\[
		\sum_i x_i y_i = \hat\beta_0\sum_i x_i + \hat\beta_1\sum_i x_i^2
	\]
	\pause
	Since $x_i\hat y_i = \hat\beta_0 x_i + \hat\beta_1 x_i^2$, summing gives $\sum_i x_i\hat y_i$ equal to the same right-hand side, so
	\[
		\sum_i x_i y_i = \sum_i x_i \hat y_i \quad\Longrightarrow\quad \boxed{\sum_i x_i\hat\varepsilon_i = 0}
	\]
\end{frame}


\begin{frame}{Property 4 \& 5: Weighted Residuals Sum to Zero (cont.)}
	For the last property, write $\hat y_i = \hat\beta_0 + \hat\beta_1 x_i$ and use both results just derived:
	\[
		\sum_i \hat y_i\hat\varepsilon_i = \hat\beta_0\underbrace{\sum_i \hat\varepsilon_i}_{=0} + \hat\beta_1\underbrace{\sum_i x_i\hat\varepsilon_i}_{=0} = 0
	\]
	\[
		\boxed{\sum_i \hat y_i \hat\varepsilon_i = 0}
	\]
\end{frame}


\section{Estimating the Error Variance \texorpdfstring{$\sigma^2$}{sigma-squared}}


\begin{frame}{Residuals and the Error Sum of Squares}
	The parameter $\sigma^2$ measures how much sales vary around the regression line for a given level of advertising spend --- it is unknown and must be estimated from the residuals.
	\svspace{0.5em}
	Define the \textbf{error sum of squares}:
	\[
		SS_E = \sum_{i=1}^n \hat\varepsilon_i^2 = \sum_{i=1}^n (y_i - \hat y_i)^2
	\]
	\pause
	It can be shown that $E[SS_E] = (n-2)\sigma^2$.
\end{frame}


\begin{frame}{An Unbiased Estimator of $\sigma^2$}
	Since $E[SS_E] = (n-2)\sigma^2$, we can write:
	\[
		\boxed{\hat\sigma^2 = MS_E = \frac{SS_E}{n-2}}
	\]
	The divisor $n-2$ reflects that two degrees of freedom were used up in estimating $\beta_0$ and $\beta_1$ before we could compute residuals.
	
	\svspace{0.5em}

	\pause

	Since $E[SS_E] = (n-2)\sigma^2$, it follows that
	\[
		E[\hat\sigma^2] = \frac{E(SS_E)}{n-2} = \frac{(n-2) \sigma^2}{n-2} = \sigma^2
	\]
	so $MS_E$ is an unbiased estimator of the error variance. This quantity is required for confidence intervals, and hypothesis tests for $\beta_0$ and $\beta_1$.
\end{frame}

\section*{Finding the Expected Value of \texorpdfstring{\(SS_E\)}{SSE}}


\begin{frame}{Why $E[SS_E] = (n-2)\sigma^2$: Rewriting the Residual}
	Recall $\hat\beta_0 = \bar y - \hat\beta_1 \bar x$. Substituting into $\hat y_i = \hat\beta_0 + \hat\beta_1 x_i$:
	\[
		\hat y_i = (\bar y - \hat\beta_1\bar x) + \hat\beta_1 x_i = \bar y + \hat\beta_1(x_i - \bar x)
	\]
	\pause
	So each residual splits into a piece explained by $x$ and a leftover piece:
	\[
		y_i - \hat y_i = (y_i - \bar y) - \hat\beta_1(x_i - \bar x)
	\]
\end{frame}
 

\begin{frame}{Why $E[SS_E] = (n-2)\sigma^2$ (cont.)}
	Squaring and summing over $i$:
	\begin{align*}
		SS_E &= \sum_i\big[(y_i-\bar y) - \hat\beta_1(x_i-\bar x)\big]^2 \\
		 &= \underbrace{\sum_i(y_i-\bar y)^2}_{S_{yy}} - 2\hat\beta_1\underbrace{\sum_i(x_i-\bar x)(y_i-\bar y)}_{S_{xy}} + \hat\beta_1^2\underbrace{\sum_i(x_i-\bar x)^2}_{S_{xx}} \\
		 &= SS_E = S_{yy} - 2\hat\beta_1 S_{xy} + \hat\beta_1^2 S_{xx}
	\end{align*}
\end{frame}


\begin{frame}{Simplifying $SS_E$}
	So far:
	\[
		SS_E = S_{yy} - 2\hat\beta_1 S_{xy} + \hat\beta_1^2 S_{xx}
	\]
	\pause
	Since $\hat\beta_1 = S_{xy}/S_{xx}$, we have $S_{xy} = \hat\beta_1 S_{xx}$, so
	\[
		\hat\beta_1^2 S_{xx} = \hat\beta_1(\hat\beta_1 S_{xx}) = \hat\beta_1 S_{xy}
	\]
	\pause
	The last two terms combine into one:
	\[
		-2\hat\beta_1 S_{xy} + \hat\beta_1 S_{xy} = -\hat\beta_1 S_{xy}
	\]
	\pause
	\[
		\boxed{SS_E = S_{yy} - \hat\beta_1 S_{xy}}
	\]
\end{frame}
 

\begin{frame}{Expectation of $S_{yy}$}
	Subtracting $\bar y = \beta_0 + \beta_1\bar x + \bar\varepsilon$ from $y_i = \beta_0+\beta_1 x_i+\varepsilon_i$:
	\[
		y_i - \bar y = \beta_1(x_i-\bar x) + (\varepsilon_i - \bar\varepsilon)
	\]
	\pause
	Squaring and summing:
	\[
		S_{yy} = \beta_1^2 S_{xx} + 2\beta_1\sum_i(x_i-\bar x)(\varepsilon_i-\bar\varepsilon) + \sum_i(\varepsilon_i-\bar\varepsilon)^2
	\]
	\pause
	The $x_i$ are fixed constants and $E[\varepsilon_i-\bar\varepsilon]=0$, so the middle term vanishes in expectation. For the last term, using $E[\varepsilon_i^2]=\sigma^2$ and $\operatorname{Var}(\bar\varepsilon)=\sigma^2/n$:
	\[
		\sum_i(\varepsilon_i-\bar\varepsilon)^2 = \sum_i\varepsilon_i^2 - n\bar\varepsilon^2
		\;\Rightarrow\;
		E\Big[\sum_i(\varepsilon_i-\bar\varepsilon)^2\Big] = n\sigma^2 - n\cdot\frac{\sigma^2}{n} = (n-1)\sigma^2
	\]
	\pause
	\[
		\boxed{E[S_{yy}] = \beta_1^2 S_{xx} + (n-1)\sigma^2}
	\]
\end{frame}
 

\begin{frame}{Expectation of $\hat\beta_1 S_{xy}$}
	Using $S_{xy} = \hat\beta_1 S_{xx}$ again:
	\[
		\hat\beta_1 S_{xy} = \hat\beta_1^2 S_{xx}
	\]
	\pause
	Since $E[\hat\beta_1] = \beta_1$ and $\operatorname{Var}(\hat\beta_1) = \sigma^2/S_{xx}$:
	\[
		E[\hat\beta_1^2] = \operatorname{Var}(\hat\beta_1) + \big(E[\hat\beta_1]\big)^2 = \frac{\sigma^2}{S_{xx}} + \beta_1^2
	\]
	\pause
	\[
		E[\hat\beta_1 S_{xy}] = S_{xx}\left(\frac{\sigma^2}{S_{xx}} + \beta_1^2\right) = \sigma^2 + \beta_1^2 S_{xx}
	\]
\end{frame}


\begin{frame}{The Final Result}
	Combining with the previous page:
	\[
		E[SS_E] = E[S_{yy}] - E[\hat\beta_1 S_{xy}] = \big[\beta_1^2 S_{xx} + (n-1)\sigma^2\big] - \big[\sigma^2 + \beta_1^2 S_{xx}\big]
	\]
	\[
		\boxed{E[SS_E] = (n-2)\sigma^2}
	\]
	The $\beta_1^2 S_{xx}$ terms cancel exactly --- that cancellation is why $SS_E$ isolates leftover error variance rather than variance explained by the true slope.
\end{frame}
